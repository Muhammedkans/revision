// --- STEP 1: STATE (DATA STORE) ---
// We represent each task as an Object: { id: number, text: string, completed: boolean }
// This allows us to track completed status and unique IDs.
let todos = JSON.parse(localStorage.getItem('advanced_todos')) || [];

// Keep track of which todo is currently being edited (null = none)
let editingTodoId = null;

// Keep track of the active filter: 'all', 'active', or 'completed'
let currentFilter = 'all';

// --- STEP 2: DOM ELEMENTS ---
const todoInput = document.querySelector('#todo-input');
const addBtn = document.querySelector('#add-btn');
const todoList = document.querySelector('#todo-list');
const itemsLeft = document.querySelector('#items-left');
const clearCompletedBtn = document.querySelector('#clear-completed-btn');
const filterBtns = document.querySelectorAll('.filter-btn');

// --- STEP 3: RENDER FUNCTION (Updates the UI based on state) ---
function render() {
    // 1. Clear the list container
    todoList.innerHTML = '';

    // 2. Filter our data based on active filter state
    let filteredTodos = todos;
    if (currentFilter === 'active') {
        filteredTodos = todos.filter(todo => !todo.completed);
    } else if (currentFilter === 'completed') {
        filteredTodos = todos.filter(todo => todo.completed);
    }

    // 3. Render items
    filteredTodos.forEach(todo => {
        const li = document.createElement('li');
        li.className = `todo-item ${todo.completed ? 'completed' : ''}`;

        // If this specific todo is being edited, show an input field instead of text
        if (todo.id === editingTodoId) {
            li.classList.add('editing');

            const editInput = document.createElement('input');
            editInput.type = 'text';
            editInput.className = 'edit-input';
            editInput.value = todo.text;

            // Handle keyboard events inside the edit input
            editInput.addEventListener('keydown', (e) => {
                if (e.key === 'Enter') {
                    saveEdit(todo.id, editInput.value);
                } else if (e.key === 'Escape') {
                    cancelEdit();
                }
            });

            // If user clicks outside the input, save the changes automatically
            editInput.addEventListener('blur', () => {
                saveEdit(todo.id, editInput.value);
            });

            li.appendChild(editInput);
            todoList.appendChild(li);

            // Automatically place cursor and focus inside the edit input
            editInput.focus();
        } else {
            // NORMAL MODE: Display Checkbox, Text, Edit & Delete buttons

            // Container for checkbox and text
            const contentDiv = document.createElement('div');
            contentDiv.className = 'todo-content';

            // Checkbox
            const checkbox = document.createElement('input');
            checkbox.type = 'checkbox';
            checkbox.className = 'todo-checkbox';
            checkbox.checked = todo.completed;
            checkbox.addEventListener('change', () => toggleTodo(todo.id));

            // Text Label
            const span = document.createElement('span');
            span.className = 'todo-text';
            span.textContent = todo.text;
            
            // Double-click to start editing directly
            span.addEventListener('dblclick', () => startEdit(todo.id));

            contentDiv.appendChild(checkbox);
            contentDiv.appendChild(span);

            // Actions container (buttons)
            const actionsDiv = document.createElement('div');
            actionsDiv.className = 'todo-actions';

            // Edit button
            const editBtn = document.createElement('button');
            editBtn.className = 'edit-btn';
            editBtn.innerHTML = '✏️';
            editBtn.title = 'Edit task';
            editBtn.addEventListener('click', () => startEdit(todo.id));

            // Delete button
            const deleteBtn = document.createElement('button');
            deleteBtn.className = 'delete-btn';
            deleteBtn.innerHTML = '&times;';
            deleteBtn.title = 'Delete task';
            deleteBtn.addEventListener('click', () => deleteTodo(todo.id));

            actionsDiv.appendChild(editBtn);
            actionsDiv.appendChild(deleteBtn);

            // Assemble list item
            li.appendChild(contentDiv);
            li.appendChild(actionsDiv);
            todoList.appendChild(li);
        }
    });

    // 4. Update the bottom counter status
    updateFooter();
}

// --- STEP 4: STATE MANAGEMENT FUNCTIONS ---

// Add a new todo item
function addTodo() {
    const text = todoInput.value.trim();
    if (text === '') return;

    // Create a new todo object with a unique timestamp ID
    const newTodo = {
        id: Date.now(),
        text: text,
        completed: false
    };

    todos.push(newTodo);
    saveToLocalStorage();
    render();
    
    // Clear main input box
    todoInput.value = '';
}

// Delete a todo item by ID
function deleteTodo(id) {
    todos = todos.filter(todo => todo.id !== id);
    saveToLocalStorage();
    render();
}

// Toggle task completion
function toggleTodo(id) {
    todos = todos.map(todo => {
        if (todo.id === id) {
            return { ...todo, completed: !todo.completed };
        }
        return todo;
    });
    saveToLocalStorage();
    render();
}

// Start inline edit mode
function startEdit(id) {
    editingTodoId = id;
    render();
}

// Save the edited text
function saveEdit(id, newText) {
    const trimmedText = newText.trim();
    
    if (trimmedText === '') {
        // If the user clears the text entirely, delete the task
        deleteTodo(id);
    } else {
        todos = todos.map(todo => {
            if (todo.id === id) {
                return { ...todo, text: trimmedText };
            }
            return todo;
        });
        editingTodoId = null;
        saveToLocalStorage();
        render();
    }
}

// Cancel edit mode
function cancelEdit() {
    editingTodoId = null;
    render();
}

// Clear all completed items
function clearCompleted() {
    todos = todos.filter(todo => !todo.completed);
    saveToLocalStorage();
    render();
}

// Update active item count and footer visibility
function updateFooter() {
    const activeCount = todos.filter(todo => !todo.completed).length;
    itemsLeft.textContent = `${activeCount} item${activeCount === 1 ? '' : 's'} left`;
}

// Save array state to local storage
function saveToLocalStorage() {
    localStorage.setItem('advanced_todos', JSON.stringify(todos));
}

// --- STEP 5: EVENT WIRING ---

// Add task on button click
addBtn.addEventListener('click', addTodo);

// Add task on Enter press in input field
todoInput.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') {
        addTodo();
    }
});

// Clear completed tasks on click
clearCompletedBtn.addEventListener('click', clearCompleted);

// Filter button clicks
filterBtns.forEach(btn => {
    btn.addEventListener('click', (e) => {
        // Remove active class from all buttons, and add it to the clicked button
        filterBtns.forEach(f => f.classList.remove('active'));
        e.target.classList.add('active');

        // Set the active filter and redraw the screen
        currentFilter = e.target.getAttribute('data-filter');
        render();
    });
});

// --- STEP 6: INITIAL LOAD ---
render();
